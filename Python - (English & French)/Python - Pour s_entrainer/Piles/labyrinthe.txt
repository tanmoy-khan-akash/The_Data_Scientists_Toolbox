def valide(i, j, n, p):
    "teste si les coordonnÃ©es de P sont valides"
    return 0 <= i < n and 0 <= j < p

def voisins(case, est_visite):
    "renvoie les voisins non visitÃ©s de P"
    n, p = len(est_visite), len(est_visite[0])
    i, j = case
    
    liste_voisins = []
    
    def ajout(i, j):
        "ajoute (i, j) Ã  liste_voisins s'il est valide et non visitÃ©"
        if valide(i, j, n, p) and not est_visite[i][j]:
            liste_voisins.append((i, j))
            
    for k in (-1, 1):
        ajout(i + k, j)
        ajout(i, j + k)
        
    return liste_voisins

def choisir(L):
    "renvoie un Ã©lÃ©ment alÃ©atoire de $L$"
    return L[randint(0, len(L) - 1)]
                  

def labyrinthe(n, p):
    "crÃ©e un labyrinthe parfait de taille n"
    est_visite = [[False for _  in range(p)] for _ in range(n)]
    precedent = [[None for _  in range(p)] for _ in range(n)]
    pile = []
    pile.append((0, 0))
    est_visite[0][0] = True # (0, 0) est le premier sommet visitÃ©
    segments = []
    # Tant qu'il y a des sommets ayant potentiellement encore des voisins non visitÃ©s
    while pile != []: 
        case = pile.pop()
        voisin = voisins(case, est_visite)
        if voisin != []: # S'il y a des voisins non visitÃ©s
            suivante = choisir(voisin) # on en choisit un alÃ©atoirement
            segments.append((case, suivante))
            i, j = suivante
            precedent[i][j] = case # case est son prÃ©dÃ©cesseur
            est_visite[i][j] = True # on le marque comme visitÃ©
            pile.append(case) # on rempile case au cas oÃ¹ il reste des voisins non visitÃ©s
            pile.append(suivante) # on empile suivante pour visiter ses voisins
    return segments, precedent

def laby(n, p, anim=False):
    "construction et rÃ©solution du labyrinthe"
    # calcul du labyrinthe
    segments, precedent = labyrinthe(n, p)

    # affichage du labyrinthe
    affiche(segments, n, p)

    # calcul du parcours
    case = (n - 1, p - 1)
    chemin = []
    while case != (0, 0):
        i, j = case
        case = precedent[i][j]
        chemin.append((case, (i, j)))

    # affichage du parcours
    if anim:
        affiche_couleur(chemin[::-1], n, p, 'r', True)
    else:
        affiche_couleur(chemin, n, p, 'r', False)
