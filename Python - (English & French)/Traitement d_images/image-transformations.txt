## Rotation

def rotation(x0, y0, theta, x, y):
    return np.array([x0,y0]) + np.array([[np.cos(theta),-np.sin(theta)],[np.sin(theta),np.cos(theta)]]).dot(np.array([x-x0,y-y0]))
        
x0, y0 = largeur//2, hauteur//2
theta = 2*np.pi/3
                                                                                                          
rot = np.zeros(image.shape)
for x in range(largeur):
    for y in range(hauteur):
        X, Y = rotation(x0, y0, theta, x, y).astype(int)
        if 0 <= X < largeur and 0 <= Y < hauteur :
            rot[y, x] = image[Y, X]
plt.imshow(rot)
plt.imsave('rot.png', rot)
