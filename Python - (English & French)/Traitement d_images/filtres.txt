image = plt.imread('Cameraman.png')

hauteur, largeur = image.shape[:2]

# Conversion en N&B si image en couleur
if len(image.shape) == 3:
    image = (image[:, :, 0] + image[:, :, 1] + image[:, :, 2]) / 3

def convol(I, K):
    "applique par convolution le filtre K sur l'image I et renvoie la nouvelle image."
    h, l = I.shape[:2]
    n, m = K.shape
    assert n == m and n % 2 == 1
    p = (n - 1) // 2
    I2 = I.copy()
    for i in range(p + 1, h - p):
        for j in range(p + 1, l - p):
            I2[i, j] = sum(I[i + k - 1, j + l - 1] * K[k, l] for k in range(n) for l in range(n))
    return I2

def contour():
    "DÃ©tection de contour"
    K = np.zeros((3, 3))
    K[1, 1] = 1
    K[0, 1] = -1
    K2 = np.zeros((3, 3))
    K2[1, 1] = 1
    K2[1, 0] = -1
    image2 = convol(image, K)
    image3 = convol(image, K)
    image4 = np.sqrt(image2**2 + image3**2)
    plt.imshow(image4, cmap='gray')
    plt.imsave('contour.png', image4, cmap='gray')
    plt.show()

def Sobel(image):
    K = np.zeros((3, 3))
    K[:, 0] = [1, 2, 1]
    K[:, 2] = [-1, -2, -1]
    K /= 4

    K2 = np.zeros((3, 3))
    K2[2, :] = [1, 2, 1]
    K2[0, :] = [-1, -2, -1]
    K2 /= 4

    image2 = np.sqrt(convol(image,K) ** 2 + convol(image,K2) ** 2)
    #return image2
    plt.imshow(image2, cmap='gray')
    plt.imsave('Sobel2.png', image2, cmap='gray')
    plt.show()
