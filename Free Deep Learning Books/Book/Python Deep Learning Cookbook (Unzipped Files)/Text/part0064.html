How to do it...

  1. Import the necessary libraries with the following code:
import numpy as np
from matplotlib import pyplot as plt
import scipy.io

from keras.models import Sequential
from keras.layers.core import Dense
from keras.optimizers import Adam
  1. Load the dataset and extract the data we need:
mat = scipy.io.loadmat('./data/train_32x32.mat')
mat = mat['X']
b, h, d, n = mat.shape
  1. Preprocess the data:
# Grayscale
img_gray = np.zeros(shape =(n, b * h))

def rgb2gray(rgb):    
    return np.dot(rgb[...,:3], [0.299, 0.587, 0.114])

for i in range(n):
    #Convert to greyscale
    img = rgb2gray(mat[:,:,:,i])
    img = img.reshape(1, 1024)
    img_gray[i,:] = img

# Normalize
X_train = img_gray/255.
  1. Next, let's define the autoencoder network architecture:
img_size = X_train.shape[1]
model = Sequential()
model.add(Dense(256, input_dim=img_size, activation='relu'))
model.add(Dense(128, activation='relu'))
model.add(Dense(64, activation='relu'))
model.add(Dense(32, activation='relu'))

model.add(Dense(64, activation='relu'))
model.add(Dense(128, activation='relu'))
model.add(Dense(256, activation='relu'))
model.add(Dense(img_size, activation='sigmoid'))

opt = Adam()
model.compile(loss='binary_crossentropy', optimizer=opt)
  1. Now, we can train the autencoder:
n_epochs = 100
batch_size = 512

model.fit(X_train, X_train, epochs=n_epochs, batch_size=batch_size, shuffle=True, validation_split=0.2)
  1. We want to see how our autoencoder performs on the training set:
   pred = model.predict(X_train)
  1. Let's plot some of the original images and their decoded versions:
n = 5
plt.figure(figsize=(15, 5))
for i in range(n):
    # display original
    ax = plt.subplot(2, n, i + 1)
    plt.imshow(mat[i].reshape(32, 32), cmap='gray')

    ax = plt.subplot(2, n, i + 1 + n)
    plt.imshow(pred[i].reshape(32, 32), cmap='gray')
plt.show()
